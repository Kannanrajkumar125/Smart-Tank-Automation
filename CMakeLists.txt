cmake_minimum_required(VERSION 3.10)
project(tank_automation CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)
find_package(Threads REQUIRED)
option(USE_MOSQUITTO "Real MQTT backend via libmosquitto" OFF)

add_executable(tank_sim src/main.cpp)
add_executable(tank_tests tests/test_main.cpp)
target_include_directories(tank_sim PRIVATE src)
target_include_directories(tank_tests PRIVATE src)
target_link_libraries(tank_sim Threads::Threads)
target_link_libraries(tank_tests Threads::Threads)
if(USE_MOSQUITTO)
  target_compile_definitions(tank_sim PRIVATE USE_MOSQUITTO)
  target_link_libraries(tank_sim mosquitto)
endif()
enable_testing()
add_test(NAME unit_and_e2e COMMAND tank_tests WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
